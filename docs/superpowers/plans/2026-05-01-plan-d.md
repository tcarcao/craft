## Task 1: Wrap resyncToTopLevel discards in SyntaxKindErrorNode (#1)

**Background:** `resyncToTopLevel()` in `internal/syntax/parser.go` (~line 1257) discards tokens until it finds a top-level keyword. The comment says "the lossless invariant is preserved" — the tokens ARE emitted into the builder — but they appear as loose siblings of the partially-formed declaration rather than grouped under an error node. Wrapping them in `SyntaxKindErrorNode` makes the tree structurally sound: tree walkers can detect the error region and skip or highlight it without inspecting individual token kinds.

## Task 2: SyntaxKindServiceField grammar nodes (#6)

**Background:** `parseServiceBody()` emits `contexts:`, `data-stores:`, `language:`, `deployment:` field tokens as flat children of `SyntaxKindServiceDecl`. There are no grammar nodes grouping a field's keyword + colon + values into a single subtree. Adding `SyntaxKindServiceField` nodes makes the tree mirror grammar structure and enables Task 3 (completion) to identify which field the cursor is in without a regex.

**Backwards-compatibility:** All existing `ServiceDecl` methods (`ContextTokens()`, `DataStoreTokens()`, `LanguageToken()`) use `scanBodyTokens()` → `node.Tokens()` which is recursive and collects tokens from inside `SyntaxKindServiceField` children. No change needed to those methods.

// parseServiceBody parses the fields inside a service { ... } block.
// Each field (keyword + colon + values) is wrapped in a SyntaxKindServiceField
// node so the tree mirrors the grammar structure.
func (p *Parser) parseServiceBody() []craft.Diagnostic {
	var diags []craft.Diagnostic

	for !p.atEOF() && p.peek().Type != lexer.TokenRBrace {
		tok := p.peek()

		// Field names are contextual keywords: contexts, data-stores, language.
		// They lex as TokenIdent (or hyphenated ident). We match on Value.
		if tok.Type != lexer.TokenIdent {
			diags = append(diags, p.diagUnexpected(tok, "field name (contexts, data-stores, language) or `}`"))
			if tok.Type == lexer.TokenLBrace {
				p.resyncToBlock()
			} else {
				p.consume()
			}
			continue
		}

		fieldName := tok.Value

		// Wrap the entire field (keyword + colon + values) in a ServiceField node.
		p.builder.StartNode(SyntaxKindServiceField)

		// Consume field keyword with its specific SyntaxKind so treeEnclosingBlock
		// can identify the field type without a regex.
		switch fieldName {
		case "contexts":
			p.consumeAs(SyntaxKindKwContexts)
		case "data-stores":
			p.consumeAs(SyntaxKindKwDataStores)
		case "language":
			p.consumeAs(SyntaxKindKwLanguage)
		case "deployment":
			p.consumeAs(SyntaxKindKwDeployment)
		default:
			p.consumeAs(SyntaxKindIdent)
		}

		if p.peek().Type != lexer.TokenColon {
			diags = append(diags, p.diagUnexpected(p.peek(), ":"))
			p.builder.FinishNode()
			continue
		}
		p.consumeAs(SyntaxKindColon)

		switch fieldName {
		case "contexts":
			p.parseIdentListWithLines()
		case "data-stores":
			p.parseIdentList()
		case "language":
			if p.peek().Type == lexer.TokenIdent {
				p.consumeAs(SyntaxKindIdent)
			} else {
				diags = append(diags, p.diagUnexpected(p.peek(), "language identifier"))
			}
		case "deployment":
			dd := p.parseDeploymentSpec()
			diags = append(diags, dd...)
		default:
			diags = append(diags, p.diagUnexpected(tok, "field name (contexts, data-stores, language) or `}`"))
			p.skipToNextField()
		}

		p.builder.FinishNode()
	}

	return diags
}
```

// ServiceField is a typed view over a SyntaxKindServiceField node — one field
// declaration inside a service body (field keyword + colon + value tokens).
// The field keyword is emitted with a specific SyntaxKind
// (SyntaxKindKwContexts, SyntaxKindKwLanguage, SyntaxKindKwDataStores, or
// SyntaxKindKwDeployment) so callers can identify the field type without regex.
type ServiceField struct{ node SyntaxNode }

// IsContexts reports whether this is a contexts: field.
func (sf ServiceField) IsContexts() bool {
	return sf.node.ChildToken(SyntaxKindKwContexts) != nil
}

// IsLanguage reports whether this is a language: field.
func (sf ServiceField) IsLanguage() bool {
	return sf.node.ChildToken(SyntaxKindKwLanguage) != nil
}

// IsDataStores reports whether this is a data-stores: field.
func (sf ServiceField) IsDataStores() bool {
	return sf.node.ChildToken(SyntaxKindKwDataStores) != nil
}

// IsDeployment reports whether this is a deployment: field.
func (sf ServiceField) IsDeployment() bool {
	return sf.node.ChildToken(SyntaxKindKwDeployment) != nil
}
```

Then add `Fields()` to `ServiceDecl` (after the existing `ContextRefs()` method):

```go
// Fields returns all field declarations inside this service body as ServiceField
// views. Each ServiceField corresponds to one SyntaxKindServiceField node.
func (s ServiceDecl) Fields() []ServiceField {
	children := s.node.ChildNodes(SyntaxKindServiceField)
	result := make([]ServiceField, len(children))
	for i, child := range children {
		result[i] = ServiceField{node: child}
	}
	return result
}
```

```go
func treeEnclosingBlock(root syntax.SyntaxNode, li green.LineIndex, cursorLine, cursorCol int) string {
	offset := li.Offset(cursorLine+1, cursorCol+1)
	node := root.NodeAt(green.TextSize(offset))
	for node != nil {
		switch node.Kind() {
		case syntax.SyntaxKindServiceField:
			// Identify the specific field by its first keyword token.
			if node.ChildToken(syntax.SyntaxKindKwContexts) != nil {
				return "service.contexts"
			}
			if node.ChildToken(syntax.SyntaxKindKwLanguage) != nil {
				return "service.language"
			}
			if node.ChildToken(syntax.SyntaxKindKwDataStores) != nil {
				return "service.data-stores"
			}
			return "service"
		case syntax.SyntaxKindServiceDecl:
			return "service"
		case syntax.SyntaxKindServicesBlock:
			return "services"
		case syntax.SyntaxKindDomainDecl:
			return "domain"
		case syntax.SyntaxKindDomainsBlock:
			return "domains"
		case syntax.SyntaxKindActorsBlock:
			return "actors"
		case syntax.SyntaxKindUseCaseDecl, syntax.SyntaxKindScenario:
			return "use_case"
		case syntax.SyntaxKindExposureDecl:
			return "expose"
		case syntax.SyntaxKindArchDecl:
			return "arch"
		case syntax.SyntaxKindFile:
			return ""
		}
		parent := node.Parent()
		if parent == nil {
			break
		}
		node = parent
	}
	return ""
}
```

```go
// detectContext determines what kind of completion to offer based on where
// the cursor sits in the grammar. It uses the syntax tree to find the nearest
// enclosing keyword; for service body fields the tree now returns a specific
// "service.X" string after Task 2 added SyntaxKindServiceField nodes.
// Expose field detection still uses a line-prefix regex (no grammar nodes yet).
func detectContext(root syntax.SyntaxNode, li green.LineIndex, lines []string, cursorLine, cursorCol int) completionContext {
	linePrefix := ""
	if cursorLine < len(lines) {
		l := lines[cursorLine]
		if cursorCol <= len(l) {
			linePrefix = l[:cursorCol]
		} else {
			linePrefix = l
		}
	}

	enclosing := treeEnclosingBlock(root, li, cursorLine, cursorCol)

	// Exposure field detection still requires line-prefix regex (no ServiceField
	// equivalent for ExposureDecl yet).
	if enclosing == "expose" {
		if field, ok := fieldBeforeCursor(linePrefix); ok {
			switch field {
			case "contexts":
				return ctxExposeContexts
			case "to":
				return ctxExposeTo
			case "through":
				return ctxExposeThrough
			}
		}
	}

	switch enclosing {
	case "":
		return ctxTopLevel
	case "service.contexts":
		return ctxServiceContexts
	case "service.language":
		return ctxServiceLang
	case "service", "services", "service.data-stores":
		return ctxServiceField
	case "domain", "domains":
		return ctxDomainBody
	case "actors":
		return ctxActorsBlock
	case "use_case":
		if isActionLine(linePrefix) {
			return ctxUseCaseAction
		}
		return ctxUseCaseTop
	case "expose":
		return ctxExposeField
	default:
		return ctxUnknown
	}
}
```
